// =======================
// ARRAY METHODS (JS)
// =======================

let arr = [1, 2, 3, 4]

// push() -> oxiriga element qo‘shadi
arr.push(5) // [1,2,3,4,5]

// pop() -> oxirgi elementni o‘chiradi
arr.pop() // [1,2,3,4]

// unshift() -> boshiga qo‘shadi
arr.unshift(0) // [0,1,2,3,4]

// shift() -> boshidagi elementni o‘chiradi
arr.shift() // [1,2,3,4]

// length -> uzunligini beradi
arr.length // 4

// slice(start,end) -> nusxa kesib oladi (original o‘zgarmaydi)
arr.slice(1,3) // [2,3]

// splice(start,count,items...) -> o‘chiradi yoki qo‘shadi (original o‘zgaradi)
arr.splice(1,2) // [1,4]

// concat() -> arraylarni qo‘shadi
arr.concat([5,6]) // [1,2,3,4,5,6]

// includes() -> bor-yo‘qligini tekshiradi
arr.includes(2) // true

// indexOf() -> indexini topadi
arr.indexOf(3) // 2

// join() -> stringga aylantiradi
arr.join("-") // "1-2-3-4"

// reverse() -> teskari qiladi
arr.reverse() // [4,3,2,1]

// sort() -> tartiblaydi (string kabi ishlaydi!)
[3,1,10].sort() // [1,10,3]

// map() -> har bir elementni o‘zgartiradi (yangi array)
arr.map(x => x * 2) // [2,4,6,8]

// filter() -> shart bo‘yicha ajratadi
arr.filter(x => x > 2) // [3,4]

// find() -> birinchi mos elementni topadi
arr.find(x => x > 2) // 3

// forEach() -> har birini aylanadi (return yo‘q)
arr.forEach(x => console.log(x))

// some() -> kamida bittasi to‘g‘rimi
arr.some(x => x > 3) // true

// every() -> hammasi to‘g‘rimi
arr.every(x => x > 0) // true

// reduce() -> bitta qiymatga yig‘adi
arr.reduce((sum,x) => sum + x, 0) // 10
